/* ═══════════════════ POMOĆNE ═══════════════════ */
const pad = n => String(n).padStart(2, '0');
const lerp = (a, b, t) => a + (b - a) * t;
const clamp01 = v => Math.min(1, Math.max(0, v));
const lerpColor = (c1, c2, t) => [
  Math.round(lerp(c1[0], c2[0], t)),
  Math.round(lerp(c1[1], c2[1], t)),
  Math.round(lerp(c1[2], c2[2], t))
];

/* ═══════════════════ NEBO ═══════════════════ */
const SKY = [
  { h: 0.0,  top: [  6, 10, 31], mid: [ 10, 16, 48], bot: [ 16, 24, 64] },
  { h: 4.5,  top: [  8, 14, 42], mid: [ 16, 26, 66], bot: [ 30, 40, 86] },
  { h: 6.0,  top: [ 40, 58,110], mid: [120,110,150], bot: [230,140, 90] },
  { h: 7.2,  top: [ 74,124,190], mid: [160,190,225], bot: [255,190,130] },
  { h: 10.0, top: [ 58,142,220], mid: [120,190,235], bot: [180,220,245] },
  { h: 13.0, top: [ 46,132,214], mid: [110,185,235], bot: [190,225,248] },
  { h: 16.0, top: [ 58,142,220], mid: [130,195,232], bot: [200,228,240] },
  { h: 18.0, top: [120,110,175], mid: [225,140,100], bot: [255,170, 90] },
  { h: 19.3, top: [ 58, 50,110], mid: [150, 80,110], bot: [220,110, 80] },
  { h: 21.0, top: [ 14, 20, 55], mid: [ 26, 32, 80], bot: [ 46, 44,100] },
  { h: 24.0, top: [  6, 10, 31], mid: [ 10, 16, 48], bot: [ 16, 24, 64] }
];

function getSky(h) {
  let a = SKY[0], b = SKY[SKY.length - 1];
  for (let i = 0; i < SKY.length - 1; i++) {
    if (h >= SKY[i].h && h <= SKY[i + 1].h) { a = SKY[i]; b = SKY[i + 1]; break; }
  }
  const t = (h - a.h) / (b.h - a.h || 1);
  return {
    top: lerpColor(a.top, b.top, t),
    mid: lerpColor(a.mid, b.mid, t),
    bot: lerpColor(a.bot, b.bot, t)
  };
}

function nightFactor(h) {
  if (h >= 19.5 || h < 5) return 1;
  if (h >= 7 && h < 17.5) return 0;
  if (h >= 5 && h < 7) return 1 - (h - 5) / 2;
  return (h - 17.5) / 2;
}

const moonEl = document.getElementById('moon');
const MOON_SIZE = 100;
const DPR = Math.min(window.devicePixelRatio || 1, 2);
moonEl.width  = MOON_SIZE * DPR;
moonEl.height = MOON_SIZE * DPR;
moonEl.style.width  = MOON_SIZE + 'px';
moonEl.style.height = MOON_SIZE + 'px';
const mctx = moonEl.getContext('2d');
mctx.scale(DPR, DPR);

const CRATERS = [
  [-0.26, -0.30, 0.16], [ 0.19, -0.14, 0.11], [-0.06,  0.23, 0.14],
  [ 0.31,  0.30, 0.09], [-0.41,  0.10, 0.08], [ 0.05, -0.46, 0.07],
  [ 0.43, -0.33, 0.06], [-0.18,  0.44, 0.07]
];

function moonPhase(date) {
  const SYNODIC = 29.530588853;
  const KNOWN_NEW = Date.UTC(2000, 0, 6, 18, 14);
  const days = (date.getTime() - KNOWN_NEW) / 86400000;
  let p = (days % SYNODIC) / SYNODIC;
  return p < 0 ? p + 1 : p;
}

function drawMoon(phase) {
  const S = MOON_SIZE, r = S / 2 - 4, cx = S / 2, cy = S / 2;
  mctx.clearRect(0, 0, S, S);

  const angle = 2 * Math.PI * phase;
  const x = Math.cos(angle);
  const rx = Math.abs(x) * r;
  const waxing = phase < 0.5;

  mctx.beginPath();
  mctx.arc(cx, cy, r, 0, Math.PI * 2);
  mctx.fillStyle = 'rgba(26,30,54,.7)';
  mctx.fill();

  mctx.beginPath();
  if (waxing) {
    mctx.arc(cx, cy, r, -Math.PI / 2, Math.PI / 2, false);
    mctx.ellipse(cx, cy, rx, r, 0, Math.PI / 2, -Math.PI / 2, x > 0);
  } else {
    mctx.arc(cx, cy, r, Math.PI / 2, -Math.PI / 2, true);
    mctx.ellipse(cx, cy, rx, r, 0, -Math.PI / 2, Math.PI / 2, x > 0);
  }
  mctx.closePath();

  const g = mctx.createRadialGradient(
    cx - r * 0.35, cy - r * 0.35, r * 0.15,
    cx, cy, r * 1.05
  );
  g.addColorStop(0,    '#fffdf4');
  g.addColorStop(0.55, '#f3edd4');
  g.addColorStop(1,    '#d3cbaa');
  mctx.fillStyle = g;
  mctx.fill();

  mctx.save();
  mctx.clip();
  mctx.fillStyle = 'rgba(140,134,108,.25)';
  for (const [ox, oy, cr] of CRATERS) {
    mctx.beginPath();
    mctx.arc(cx + ox * r, cy + oy * r, cr * r, 0, Math.PI * 2);
    mctx.fill();
  }
  mctx.restore();

  const k = (1 - x) / 2;
  moonEl.style.filter =
    `drop-shadow(0 0 ${(5 + 16 * k).toFixed(0)}px ` +
    rgba(244,240,214,${(0.08 + 0.5 * k).toFixed(2)}));
}

function phaseName(p) {
  if (p < 0.03 || p > 0.97) return 'mlad mesec';
  if (p < 0.22) return 'rastući srp';
  if (p < 0.28) return 'prva četvrt';
  if (p < 0.47) return 'rastući mesec';
  if (p < 0.53) return 'pun mesec';
  if (p < 0.72) return 'opadajući mesec';
  if (p < 0.78) return 'poslednja četvrt';
  return 'opadajući srp';
}

/* ═══════════════════ 13-MESEČNI KALENDAR ═══════════════════ */
const MONTHS = [
  'Januar', 'Februar', 'Mart', 'April',
  'Maj', 'Jun', 'Jul', 'Avgust',
  'Septembar', 'Oktobar', 'Novembar', 'Decembar',
  'Vinčanac'
];
const MONTHS_ABBR = ['Jan','Feb','Mar','Apr','Maj','Jun','Jul','Avg','Sep','Okt','Nov','Dec','Vin'];
const WEEKDAYS = ['P','U','S','Č','P','S','N'];

function isLeapYear(y) {
  return (y % 4 === 0 && y % 100 !== 0) || (y % 400 === 0);
}

function toNewCalendar(d) {
  const y = d.getFullYear();
  const yStart = new Date(y, 0, 1);
  const doy = Math.floor((d - yStart) / 86400000) + 1;
  const leap = isLeapYear(y);
  const total = leap ? 366 : 365;

  if (doy >= 1 && doy <= 364) {
    return {
      month: Math.floor((doy - 1) / 28),
      day: ((doy - 1) % 28) + 1,
      year: y, special: null, doy, total, leap
    };
  }
  if (doy === 365) {
    return { month: null, day: null, year: y, special: 'Dan godine', doy, total, leap };
  }
  if (doy === 366 && leap) {
    return { month: null, day: null, year: y, special: 'Dan prestupa', doy, total, leap };
  }
  return { month: null, day: null, year: y, special: '?', doy, total, leap };
}

function partOfDay(h) {
  if (h < 5)    return 'noć';
  if (h < 7)    return 'zora';
  if (h < 11)   return 'jutro';
  if (h < 14)   return 'podne';
  if (h < 17)   return 'popodne';
  if (h < 19.5) return 'sumrak';
  if (h < 22)   return 'veče';
  return 'noć';
}

let lastPhaseKey = '';

/* ═══════════════════ GLAVNI TIK ═══════════════════ */
function tick() {
  const now = new Date();
  const h = now.getHours() + now.getMinutes() / 60 + now.getSeconds() / 3600;

  /* nebo */
  const s = getSky(h);
  skyEl.style.background =
    `linear-gradient(to bottom, rgb(${s.top.join(',')}) 0%, ` +
    rgb(${s.mid.join(',')}) 52%, rgb(${s.bot.join(',')}) 100%);

  /* zvezde */
  starsEl.style.opacity = nightFactor(h).toFixed(3);

  /* sunce */
  const ts = (h - 6) / 12;
  const thS = Math.PI * ts;
  sunEl.style.left = (50 - 40 * Math.cos(thS)).toFixed(2) + '%';
  sunEl.style.top  = (48 - 33 * Math.sin(thS)).toFixed(2) + '%';
  sunEl.style.opacity = (ts > -0.15 && ts < 1.15) ? '1' : '0';

  const elev = Math.max(0, Math.sin(thS));
  const warm = clamp01(1 - elev * 2.2);
  const c1 = lerpColor([255, 229, 138], [255, 150,  70], warm);
  const c2 = lerpColor([255, 179,  71], [255, 105,  40], warm);
  sunEl.style.background =
    radial-gradient(circle at 38% 34%, #fffdf2, rgb(${c1.join(',')}) 48%, rgb(${c2.join(',')}) 100%);
  sunEl.style.boxShadow =
    0 0 28px 8px rgba(255,${Math.round(214 - 95 * warm)},${Math.round(120 - 70 * warm)},.6), +
    0 0 90px 30px rgba(255,${Math.round(170 - 75 * warm)},${Math.round(70 - 35 * warm)},.28);

  /* mesec */
  const tm = (((h - 18 + 24) % 24)) / 12;
  if (tm >= 0 && tm <= 1) {
    const thM = Math.PI * tm;
    moonEl.style.left = (50 - 40 * Math.cos(thM)).toFixed(2) + '%';
    moonEl.style.top  = (48 - 33 * Math.sin(thM)).toFixed(2) + '%';
    moonEl.style.opacity = '1';
  } else {
    moonEl.style.opacity = '0';
  }

  const phase = moonPhase(now);
  const phaseKey = now.toDateString();
  if (phaseKey !== lastPhaseKey) {
    drawMoon(phase);
    lastPhaseKey = phaseKey;
  }

  /* kalendar */
  const cal = toNewCalendar(now);

  timeEl.textContent = ${pad(now.getHours())}:${pad(now.getMinutes())}:${pad(now.getSeconds())};
  partEl.textContent = partOfDay(h);

  if (cal.special) {
    dateEl.innerHTML = <span class="special-name">${cal.special}</span> &nbsp;·&nbsp; Godina ${cal.year};
  } else {
    dateEl.innerHTML =
      Godina ${cal.year} &nbsp;·&nbsp; <b>${MONTHS[cal.month]}</b>, ${cal.day}. dan;
  }

  const chips = monthsRow.children;
  for (let i = 0; i < chips.length; i++) {
    chips[i].classList.toggle('active', cal.month === i);
  }

  if (cal.special) {
    daysGrid.innerHTML = <div class="special-banner">${cal.special}</div>;
  } else {
    const jan1 = new Date(cal.year, 0, 1).getDay();
    const offset = (jan1 + 6) % 7;
    const cells = [];
    for (let i = 0; i < offset; i++) cells.push(null);
    for (let d = 1; d <= 28; d++) cells.push(d);
    while (cells.length % 7 !== 0) cells.push(null);

    daysGrid.innerHTML = cells.map(d => {
      if (d === null) return '<span class="day empty"></span>';
      const active = (d === cal.day) ? ' active' : '';
      return <span class="day${active}">${d}</span>;
    }).join('');
  }

  const prog = cal.doy / cal.total;
  fillEl.style.width = (prog * 100).toFixed(2) + '%';

  subEl.textContent =
    ${phaseName(phase)} · dan ${cal.doy}. od ${cal.total} · ${Math.round(prog * 100)}% godine;
}

tick();
setInterval(tick, 200);
